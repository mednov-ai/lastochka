def nodes(p):
 s=p.get('Node Type','') + (' '+p['Index Name'] if 'Index Name' in p else '')
 return [s]+sum((nodes(c) for c in p.get('Plans',[])),[])
def phase(name):
 print('\nPHASE',name,flush=True)
 for key,query in queries.items():
  data=[]
  for i in range(7):
   out=run('BEGIN; EXPLAIN (ANALYZE,BUFFERS,FORMAT JSON,TIMING OFF) '+query+'; ROLLBACK;')
   obj=json.loads(out)[0]; data.append(obj)
  times=[x['Execution Time'] for x in data]
  p=data[-1]['Plan']
  print(json.dumps({'query':key,'median_ms':round(statistics.median(times),3),'min_ms':round(min(times),3),'max_ms':round(max(times),3),'shared_hits':p.get('Shared Hit Blocks'),'shared_reads':p.get('Shared Read Blocks'),'nodes':nodes(p)}),flush=True)
phase('before')
run("""CREATE INDEX audit_payment_order ON payment_requests(created_at DESC,id DESC);
CREATE INDEX audit_telegram_order ON telegram_outbox(created_at,enquiry_id) WHERE status IN ('PENDING','RETRY_WAIT');
CREATE INDEX audit_push_order ON browser_push_outbox(created_at,enquiry_id) WHERE status IN ('PENDING','RETRY_WAIT');
CREATE INDEX audit_alfa_due ON alfa_test_attempts(environment,next_check_at,payment_id) WHERE ((archived=false AND operation_state NOT IN ('PAID','CANCELLED')) OR callback_pending=true);
CREATE INDEX audit_session_end ON studio_sessions(ends_at);
ANALYZE;""")
phase('after')
print('\nINDEX_SIZES',run("SELECT relname,pg_size_pretty(pg_relation_size(oid)) FROM pg_class WHERE relname LIKE 'audit_%' ORDER BY relname"),flush=True)
